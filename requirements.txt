    <script>
        function submitQuery() {
            const query = document.getElementById('queryInput').value.trim();
            if (!query) {
                showError('Query cannot be empty.');
                return;
            }

            const showAll = document.getElementById('showAllCheckbox').checked;
            submitToAPI(query, showAll);
        }

        function submitToAPI(query, showAll) {
            const submitBtn = document.getElementById('submitBtn');
            const loading = document.getElementById('loading');
            const errorBox = document.getElementById('errorBox');
            const answerSection = document.getElementById('answerSection');
            const detailsSection = document.getElementById('detailsSection');

            // Reset UI
            errorBox.classList.remove('visible');
            answerSection.classList.remove('visible');
            detailsSection.style.display = 'none';
            submitBtn.disabled = true;
            loading.classList.add('active');

            fetch('/api/query', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify({
                    query: query,
                    show_all: showAll,
                }),
            })
                .then((response) => response.json())
                .then((data) => {
                    loading.classList.remove('active');
                    submitBtn.disabled = false;

                    if (data.error) {
                        showError(data.error);
                    } else {
                        displayAnswer(data, showAll);
                    }
                })
                .catch((error) => {
                    loading.classList.remove('active');
                    submitBtn.disabled = false;
                    showError(`Error: ${error.message}`);
                });
        }

        function showError(message) {
            const errorBox = document.getElementById('errorBox');
            errorBox.textContent = message;
            errorBox.classList.add('visible');
        }

        function displayAnswer(data, showAll) {
            const answerText = document.getElementById('answerText');
            const answerSection = document.getElementById('answerSection');
            const detailsSection = document.getElementById('detailsSection');

            answerText.textContent = data.answer;
            answerSection.classList.add('visible');

            if (showAll && data.details) {
                document.getElementById('planDetail').textContent = data.details.plan;
                document.getElementById('researchDetail').textContent = data.details.research;
                document.getElementById('verificationDetail').textContent = data.details.verification;
                document.getElementById('skepticDetail').textContent = data.details.skeptic;
                detailsSection.style.display = 'grid';
            }
        }

        // Allow Enter key to submit
        document.getElementById('queryInput').addEventListener('keypress', (e) => {
            if (e.key === 'Enter' && !e.shiftKey) {
                e.preventDefault();
                submitQuery();
            }
        });
    </script>
